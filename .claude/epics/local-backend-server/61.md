## Implementation Details

### Configuration Structure
```go
type Config struct {
    Claude struct {
        APIKey     string `mapstructure:"claude_api_key" validate:"required"`
        Model      string `mapstructure:"claude_model" validate:"required"`
        MaxTokens  int    `mapstructure:"claude_max_tokens" validate:"min=100"`
    }
    RabbitMQ struct {
        URL          string `mapstructure:"rabbitmq_url" validate:"required"`
        QueueName    string `mapstructure:"queue_name" validate:"required"`
        MaxRetries   int    `mapstructure:"max_retries" validate:"min=0"`
    }
    Database struct {
        Path        string `mapstructure:"database_path" validate:"required"`
        MaxConn     int    `mapstructure:"db_max_connections" validate:"min=1"`
    }
    Logging struct {
        Level       string `mapstructure:"log_level" validate:"required"`
        Format      string `mapstructure:"log_format" validate:"oneof=json console"`
    }
    Server struct {
        HealthPort  int    `mapstructure:"health_check_port" validate:"min=1024"`
    }
}
```
